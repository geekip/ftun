include("${PROJECT_SOURCE_DIR}/cmake/FtunLayers.cmake")
set(ftun_module_definitions)
foreach(module PROXY DNS FAKEIP TUN FRS)
  if(FTUN_MODULE_${module} STREQUAL "BUILTIN")
    list(APPEND ftun_module_definitions FTUN_HAVE_MODULE_${module}=1)
  endif()
endforeach()
add_subdirectory(util)
add_subdirectory(platform)
add_subdirectory(core)
add_subdirectory(libs)
# 配置读写独立于业务运行目标，裁剪构建仍保留旧字段的校验语义。
add_subdirectory(dns/config)
add_subdirectory(proxy/config)
if(FTUN_MODULE_PROXY STREQUAL "BUILTIN")
  add_subdirectory(proxy/protocol)
endif()
if(FTUN_MODULE_DNS STREQUAL "BUILTIN")
  add_subdirectory(dns)
endif()
if(FTUN_MODULE_PROXY STREQUAL "BUILTIN")
  add_subdirectory(proxy)
endif()
add_subdirectory(app)
add_library(ftun_runtime INTERFACE)
target_include_directories(ftun_runtime INTERFACE
  "$<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(ftun_runtime INTERFACE ftun_app)
set(ftun_layer_targets ftun_util ftun_platform ftun_core
  ftun_lib_buffer ftun_lib_transport ftun_lib_resolver ftun_lib_tun_config ftun_dns_config ftun_proxy_config ftun_app_config ftun_app)
foreach(target ftun_dns ftun_proxy ftun_proxy_protocol ftun_lib_tun)
  if(TARGET ${target})
    list(APPEND ftun_layer_targets ${target})
  endif()
endforeach()
if(FTUN_MODULE_PROXY STREQUAL "BUILTIN")
  list(APPEND ftun_layer_targets ftun_lib_tls ftun_lib_http)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ftun-build-modules.txt" CONTENT
  "core=BUILTIN\nproxy=${FTUN_MODULE_PROXY}\ndns=${FTUN_MODULE_DNS}\nfakeip=${FTUN_MODULE_FAKEIP}\ntun=${FTUN_MODULE_TUN}\nfrs=${FTUN_MODULE_FRS}\nmesh=OFF\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ftun-build-targets.txt" CONTENT
  "core=ftun_core\nutil=ftun_util\nplatform=ftun_platform\nproxy=$<IF:$<TARGET_EXISTS:ftun_proxy>,ftun_proxy,OFF>\napp=ftun_app\ntun=$<IF:$<TARGET_EXISTS:ftun_lib_tun>,ftun_lib_tun,OFF>\nlwip=$<IF:$<TARGET_EXISTS:ftun_lwip>,ftun_lwip,OFF>\ndns=$<IF:$<TARGET_EXISTS:ftun_dns>,ftun_dns,OFF>\nfakeip=$<IF:$<STREQUAL:${FTUN_MODULE_FAKEIP},BUILTIN>,ftun_proxy,OFF>\nfrs=$<IF:$<STREQUAL:${FTUN_MODULE_FRS},BUILTIN>,ftun_proxy,OFF>\nmesh=OFF\n")
set(ftun_layer_targets ${ftun_layer_targets} PARENT_SCOPE)
