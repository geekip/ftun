cmake_minimum_required(VERSION 3.21)
project(FTUN VERSION 1.0.5 LANGUAGES C)
include(cmake/FtunOptions.cmake)
# 单独编译公共层，供嵌入式集成检查使用。
option(FTUN_PLATFORM_COMPILE_CHECK "Compile util, platform, core and libs without a native application" OFF)
if(FTUN_PLATFORM_COMPILE_CHECK)
  include(cmake/FtunDependencies.cmake)
  include(cmake/FtunLayers.cmake)
  add_subdirectory(src/util)
  add_subdirectory(src/platform)
  add_subdirectory(src/core)
  add_subdirectory(src/libs)
  add_custom_target(ftun_compile_check DEPENDS ftun_util ftun_platform ftun_core
    ftun_lib_buffer ftun_lib_transport ftun_lib_resolver ftun_lib_tls ftun_lib_http
    ftun_lib_tun_config)
  return()
endif()
include(cmake/FtunDependencies.cmake)
add_subdirectory(src)
include(cmake/FtunSchema.cmake)
include(cmake/FtunInstall.cmake)
include(tests/CMakeLists.txt)

option(FTUN_PORTABLE_TESTS "Build native platform contracts without Linux-only test fixtures" OFF)
if(FTUN_PORTABLE_TESTS)
  add_executable(ftun-platform-test tests/portable/platform.c)
  target_include_directories(ftun-platform-test PRIVATE src)
  target_link_libraries(ftun-platform-test PRIVATE ftun_runtime ftun_options)
  if(WIN32)
    if(FTUN_MODULE_TUN STREQUAL "BUILTIN")
      add_executable(ftun-wintun-driver-test tests/portable/wintun_driver.c)
      target_include_directories(ftun-wintun-driver-test PRIVATE src)
      target_link_libraries(ftun-wintun-driver-test PRIVATE ftun_runtime ftun_options)
      add_executable(ftun-wintun-test tests/portable/wintun.c)
      target_include_directories(ftun-wintun-test PRIVATE src)
      target_include_directories(ftun-wintun-test SYSTEM PRIVATE third_party/wintun/api)
      target_link_libraries(ftun-wintun-test PRIVATE ftun_runtime ftun_options)
    endif()
    add_executable(ftun-wepoll-test tests/portable/wepoll.c)
    target_include_directories(ftun-wepoll-test PRIVATE src)
    target_link_libraries(ftun-wepoll-test PRIVATE ftun_runtime ftun_options)
  endif()
  if(FTUN_MODULE_PROXY STREQUAL "BUILTIN")
    # 复用既有内存TLS正反证书契约；相对fixture路径可在交叉目标上执行。
    add_executable(ftun-platform-tls-test tests/unit/libs/tls/test_tls.c)
    target_include_directories(ftun-platform-tls-test PRIVATE src)
    target_compile_definitions(ftun-platform-tls-test PRIVATE
      FTUN_TEST_CERT_DIR=third_party/wolfssl/certs)
    target_link_libraries(ftun-platform-tls-test PRIVATE ftun_runtime ftun_options wolfssl::wolfssl)
  endif()
endif()
